cmake_minimum_required(VERSION 3.19)

if(NOT DEFINED ENV{ESP_MATTER_PATH})
    message(FATAL_ERROR "Run uv run fw build (see README.md)")
endif()
if(NOT DEFINED IDF_TARGET)
    set(IDF_TARGET esp32c3)
endif()
if(NOT DEFINED FIRMWARE_TARGET)
    set(FIRMWARE_TARGET esp32-c3)
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/targets.json" firmware_targets)
string(JSON firmware_chip GET "${firmware_targets}" "${FIRMWARE_TARGET}" idf_target)
if(NOT IDF_TARGET STREQUAL firmware_chip)
    message(FATAL_ERROR "${FIRMWARE_TARGET} requires IDF_TARGET=${firmware_chip}")
endif()
if(DEFINED DEYE_CONFIGURED_TARGET AND NOT DEYE_CONFIGURED_TARGET STREQUAL FIRMWARE_TARGET)
    message(FATAL_ERROR "Build directory belongs to ${DEYE_CONFIGURED_TARGET}; use a separate directory for ${FIRMWARE_TARGET}")
endif()
set(DEYE_CONFIGURED_TARGET "${FIRMWARE_TARGET}" CACHE INTERNAL "Firmware variant for this build directory")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.${FIRMWARE_TARGET}")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.${FIRMWARE_TARGET}")
endif()
set(PROJECT_VER "0.1.0")
set(PROJECT_VER_NUMBER 1)
file(TO_CMAKE_PATH "$ENV{ESP_MATTER_PATH}" ESP_MATTER_PATH)
set(MATTER_SDK_PATH "${ESP_MATTER_PATH}/connectedhomeip/connectedhomeip")
# ESP Insights is pulled in by CHIP's manifest and omits this dependency itself.
set(COMPONENTS main esp_http_client)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# ESP Insights (a CHIP dependency) queries git_describe even in a new repository.
# Its bundled helper cannot read an unborn HEAD; use the release version then.
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --verify HEAD
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}"
    RESULT_VARIABLE git_head_result OUTPUT_QUIET ERROR_QUIET)
if(NOT git_head_result EQUAL 0)
    function(git_describe output_variable)
        set(${output_variable} "${PROJECT_VER}" PARENT_SCOPE)
    endfunction()
endif()
set(EXTRA_COMPONENT_DIRS
    "${MATTER_SDK_PATH}/config/esp32/components"
    "${ESP_MATTER_PATH}/components")
# Board variants share an IDF chip but have distinct firmware and model names.
set(FIRMWARE_NAME "${FIRMWARE_TARGET}")
project(${FIRMWARE_NAME})
idf_build_set_property(CXX_COMPILE_OPTIONS "-std=gnu++17;-Os;-DCHIP_HAVE_CONFIG_H;-Wno-overloaded-virtual" APPEND)
idf_build_set_property(COMPILE_OPTIONS "-Wno-format-nonliteral;-Wno-format-security" APPEND)
